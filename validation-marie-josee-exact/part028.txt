function contributionSummary(member){
  const all=Array.isArray(member?.contributionPayments)?member.contributionPayments:[];
  const plan=member?.contributionPlan||(all[0]?.plan)||'';
  const same=plan?all.filter(p=>p.plan===plan):[];
  const confirmed=same.filter(p=>p.status==='Confirmé');
  const pending=same.find(p=>p.status!=='Confirmé');
  const max=plan==='two'?2:1;
  const interval=plan==='two'?6:12;
  let label='Non commencée',next='—';
  if(plan){label=confirmed.length+'/'+max+' paiement'+(confirmed.length>1?'s':'')+' confirmé'+(confirmed.length>1?'s':'');}
  if(pending){label+=(confirmed.length?' · ':'')+'1 à vérifier';next='En attente de validation Direction';}
  else if(plan&&confirmed.length>=max){next='Cotisation annuelle à jour';}
  else if(plan&&confirmed.length){const last=confirmed[confirmed.length-1];next='Au plus tard le '+formatDateFr(addMonthsIso(last.date,interval));}
  else if(plan){next='À démarrer';}
  return {plan,confirmed:confirmed.length,max,label,next};
}

function withTimeout(promise,ms,message){

function fileToDataUrl(file){
  return new Promise((resolve,reject)=>{
    const r=new FileReader();
    r.onload=()=>resolve(String(r.result||''));
    r.onerror=()=>reject(new Error('Lecture du fichier impossible.'));
    r.readAsDataURL(file);
  });
}
async function openProofRecord(proof){

  try{
    const res=await fetch(proof.dataUrl);
    const blob=await res.blob();
    const url=URL.createObjectURL(blob);
    window.open(url,'_blank','noopener');
    setTimeout(()=>URL.revokeObjectURL(url),120000);
  }catch(err){
    console.error(err);
    alert('Le justificatif ne peut pas être ouvert.');
  }
}

function addMonthsIso(dateStr,months){
  if(!dateStr)return'';
  const d=new Date(dateStr+'T12:00:00Z');
  const day=d.getUTCDate();
  const tm=d.getUTCMonth()+months;
  const y=d.getUTCFullYear()+Math.floor(tm/12);
  const m=((tm%12)+12)%12;
  const last=new Date(Date.UTC(y,m+1,0)).getUTCDate();
  return new Date(Date.UTC(y,m,Math.min(day,last))).toISOString().slice(0,10);
}

}

function refreshContributionForm(){
  if(!currentMember)return;
  normalizeContributionState();
  const plan=document.querySelector('input[name=contributionPlan]:checked')?.value||currentMember.contributionPlan||'monthly';
  currentMember.contributionPlan=plan;
  const payments=currentMember.contributionPayments||[];
  const same=payments.filter(p=>p.plan===plan);
  const nextSequence=same.length+1;
  const box=document.getElementById('contributionNextBox');
  if(!box)return;

  const pending=same.find(p=>p.status!=='Confirmé');
  if(pending){
    box.innerHTML=`<strong>${contributionPaymentLabel(pending)} déjà déclaré - ${pending.amount} €.</strong><br>Statut : en attente de confirmation par la Direction.`;
    return;
  }
  if(plan==='monthly'){
    if(nextSequence>1){
      box.innerHTML='<strong>Cotisation annuelle complète.</strong><br>Le paiement annuel a été confirmé.';
      return;
    }
    box.innerHTML=`<strong>Cotisation annuelle : ${settings.monthlyContribution*12} € en une seule fois</strong><br>Base de calcul : ${settings.monthlyContribution} € × 12.<br>Après déclaration, la Direction vérifiera la réception.`;
  }else{
    if(nextSequence>2){
      box.innerHTML='<strong>Cotisation annuelle complète.</strong><br>Les deux versements ont été confirmés.';
      return;
    }
    const first=same[0];
    const due=nextSequence===2&&first?.date?addMonthsIso(first.date,6):'';
    box.innerHTML=`<strong>${nextSequence===1?'Premier':'Deuxième'} versement : ${settings.monthlyContribution*6} €</strong>${due?'<br>À régler au plus tard le '+formatDateFr(due):''}<br>Après déclaration, la Direction vérifiera la réception.`;
  }
}
async function declareContributionPayment(){
  if(!currentMember?.active||!currentMember?.paymentConfirmed){
    alert('Votre adhésion doit d’abord être validée par la Direction.');
    return;
  }
  normalizeContributionState();
  const plan=document.querySelector('input[name=contributionPlan]:checked')?.value||'monthly';
  const reference=val('contributionRef');
  const date=val('contributionDate');
  const method=document.querySelector('input[name=contributionMethod]:checked')?.value||'Airtel Money';
  if(!reference||!date){alert('Merci de renseigner la référence et la date du paiement.');return}

  const same=currentMember.contributionPayments.filter(p=>p.plan===plan);
  if(same.some(p=>p.status!=='Confirmé')){alert('Le paiement précédent doit d’abord être confirmé par la Direction.');return}
  const sequence=same.length+1;
  const max=plan==='monthly'?1:2;
  if(sequence>max){alert('Toutes les échéances de ce plan ont déjà été confirmées.');return}
  const amount=plan==='monthly'?settings.monthlyContribution*12:settings.monthlyContribution*6;
  const id='COT-'+Date.now();
  const file=document.getElementById('contributionProof')?.files?.[0]||null;
  let proof=null;
  if(file){try{proof=await uploadProofFile(file,'contributions',id)}catch(err){console.error(err);alert('Le justificatif n’a pas pu être archivé : '+(err.message||err));return}}
  const payment={
    id,plan,sequence,amount,date,method,reference,proof,
    status:'À vérifier',
    dueDate:plan==='two'&&sequence===1?addMonthsIso(date,6):''
  };
  currentMember.contributionPlan=plan;
  currentMember.contributionPayments.push(payment);
  syncMemberEverywhere(currentMember);
  persist();
  await saveMemberToFirebase(currentMember).catch(console.error);
  setVal('contributionRef','');
  setVal('contributionDate','');
  const inp=document.getElementById('contributionProof');if(inp)inp.value='';
  renderContributionMember();
  renderMemberSituation();
  refreshDirection();
  alert('Cotisation déclarée. Elle est maintenant visible par la Direction pour confirmation.'+(proof?' Le justificatif est archivé dans votre dossier.':''));
}

function confirmContributionPaymentByUid(uid,paymentId){
  const i=applications.findIndex(a=>a.firebaseUid===uid);
  if(i<0)return;
  const member=applications[i];
  const p=(member.contributionPayments||[]).find(x=>x.id===paymentId);
  if(!p)return;
  p.status='Confirmé';
  p.confirmedAt=new Date().toISOString();
  applications[i]=member;
  const ai=activeMembers.findIndex(a=>a.firebaseUid===uid);
  if(ai>=0)activeMembers[ai]=JSON.parse(JSON.stringify(member));
  if(currentMember?.firebaseUid===uid)currentMember=JSON.parse(JSON.stringify(member));
  persist();
  saveAnyMemberToFirebase(member).catch(console.error);
  renderContributionPayments();
  renderActive();
  if(currentMember?.firebaseUid===uid){renderContributionMember();renderMemberSituation();}
  alert('Cotisation confirmée par la Direction.');
}
function confirmContributionPayment(memberId,paymentId){
  const member=applications.find(a=>a.id===memberId);
  if(member?.firebaseUid)confirmContributionPaymentByUid(member.firebaseUid,paymentId);
}
