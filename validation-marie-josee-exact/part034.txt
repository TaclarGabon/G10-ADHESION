  const member=findMemberByFirebaseUid(uid);
  if(!member)return;
  const name=[member.firstName,member.firstName2,member.lastName,member.lastName2].filter(Boolean).join(' ')||member.email||member.id||'cet adhérent';
  const date=member.membershipValidatedAt?formatDateFr(member.membershipValidatedAt.slice(0,10)):'date inconnue';
  const dup=isLogicalDuplicate(member);
  const msg=(dup?'Un doublon a été détecté.\n\n':'')+
    'Supprimer cette ligne d’adhérent actif ?\n'+name+'\nValidation : '+date+
    '\n\nCette action supprime uniquement ce dossier Firebase précis.';
  if(!confirm(msg))return;
  try{
    if(db&&uid)await db.collection('members').doc(uid).delete();
    applications=applications.filter(x=>x.firebaseUid!==uid);
    activeMembers=activeMembers.filter(x=>x.firebaseUid!==uid);
    const stillActive=applications.some(x=>x.active&&(
      (member.id&&x.id===member.id) ||
      (member.email&&normalizeEmail(x.email)===normalizeEmail(member.email))
    ));
    if(!stillActive&&member.platformIndex!==undefined&&platform[member.platformIndex]){
      platform[member.platformIndex].status='Non actif';
      savePlatformEntryToFirebase(member.platformIndex).catch(console.error);
    }
    persist();
    await updatePublicSummary().catch(console.error);
    refreshDirection();renderActive();renderHistory();refreshLanding();
    alert('La ligne sélectionnée a été supprimée.');
  }catch(err){
    console.error(err);
    alert('Suppression impossible : '+(err.message||err));
  }
}
